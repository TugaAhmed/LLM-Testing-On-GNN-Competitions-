import pandas as pd
import torch
import torch.nn.functional as F
from torch_geometric.data import Data
from torch_geometric.loader import DataLoader
from torch_geometric.nn import GCNConv, global_mean_pool
from sklearn.model_selection import train_test_split
from sklearn.metrics import f1_score
import numpy as np
import os
import random

# Set seeds for reproducibility
seed_value = 42
np.random.seed(seed_value)
torch.manual_seed(seed_value)
random.seed(seed_value)

# --- 1. Load Graphs from CSVs ---
def load_graphs_from_csv(nodes_path, edges_path, labels_path=None):
    """Loads graph data from node and edge CSV files."""
    nodes_df = pd.read_csv(nodes_path)
    edges_df = pd.read_csv(edges_path)
    
    data_list = []
    graph_labels = {}
    
    if labels_path:
        labels_df = pd.read_csv(labels_path)
        graph_labels = dict(zip(labels_df['id'], labels_df['target']))

    graph_ids = nodes_df['graph_id'].unique()

    for graph_id in graph_ids:
        graph_nodes = nodes_df[nodes_df['graph_id'] == graph_id]
        graph_edges = edges_df[edges_df['graph_id'] == graph_id]

        node_features = graph_nodes.drop(columns=['graph_id', 'node_id']).values
        x = torch.tensor(node_features, dtype=torch.float32)

        if not graph_edges.empty:
            edge_index = torch.tensor(graph_edges[['src', 'dst']].values, dtype=torch.long).t().contiguous()
            edge_attr = torch.tensor(graph_edges[['ef_0']].values, dtype=torch.float32)
        else:
            edge_index = torch.empty((2, 0), dtype=torch.long)
            edge_attr = torch.empty((0, 1), dtype=torch.float32)

        data = Data(x=x, edge_index=edge_index, edge_attr=edge_attr)
        
        if graph_id in graph_labels:
            data.y = torch.tensor([graph_labels[graph_id]], dtype=torch.long)
        
        data.graph_id = graph_id
        data_list.append(data)
        
    return data_list

# Load training and test graphs with corrected paths
all_train_graphs = load_graphs_from_csv(
        './data/public/train_nodes.csv',
        './data/public/train_edges.csv',
        './data/public/train_labels.csv')

test_graphs = load_graphs_from_csv(
        './data/public/test_nodes.csv',
        './data/public/test_edges.csv')

# Split training data into training and validation sets
train_graphs, val_graphs = train_test_split(all_train_graphs, test_size=0.2, random_state=seed_value)

# Create DataLoaders
train_loader = DataLoader(train_graphs, batch_size=32, shuffle=True)
val_loader = DataLoader(val_graphs, batch_size=32, shuffle=False)
test_loader = DataLoader(test_graphs, batch_size=32, shuffle=False)

# --- 2. Define the GNN Model ---
class GNNModel(nn.Module):
    def __init__(self, num_node_features, hidden_channels, out_channels):
        super(GNNModel, self).__init__()
        self.conv1 = GCNConv(num_node_features, hidden_channels)
        self.conv2 = GCNConv(hidden_channels, hidden_channels)
        self.conv3 = GCNConv(hidden_channels, hidden_channels)
        self.lin = nn.Linear(hidden_channels, out_channels)

    def forward(self, data):
        x, edge_index, batch = data.x, data.edge_index, data.batch
        x = self.conv1(x, edge_index)
        x = F.relu(x)
        x = F.dropout(x, p=0.5, training=self.training)
        x = self.conv2(x, edge_index)
        x = F.relu(x)
        x = F.dropout(x, p=0.5, training=self.training)
        x = self.conv3(x, edge_index)
        x = F.relu(x)
        x = global_mean_pool(x, batch)
        x = self.lin(x)
        return F.log_softmax(x, dim=1)

# Initialize model, loss function, and optimizer
num_node_features = 8
hidden_channels = 64
out_channels = 2
model = GNNModel(num_node_features, hidden_channels, out_channels)
optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
criterion = nn.NLLLoss()

# --- 3. Training Loop ---
def train():
    model.train()
    for data in train_loader:
        optimizer.zero_grad()
        out = model(data)
        loss = criterion(out, data.y)
        loss.backward()
        optimizer.step()

# --- 4. Validation Loop ---
def validate():
    model.eval()
    preds = []
    true_labels = []
    with torch.no_grad():
        for data in val_loader:
            out = model(data)
            pred = out.argmax(dim=1)
            preds.extend(pred.numpy())
            true_labels.extend(data.y.numpy())
    macro_f1 = f1_score(true_labels, preds, average='macro')
    return macro_f1

# --- 5. Main Training Routine ---
best_val_macro_f1 = 0
patience = 10
trigger_times = 0

for epoch in range(1, 101):
    train()
    val_macro_f1 = validate()
    print(f'Epoch {epoch:03d}, Val Macro F1: {val_macro_f1:.4f}')
    
    if val_macro_f1 > best_val_macro_f1:
        best_val_macro_f1 = val_macro_f1
        trigger_times = 0
        torch.save(model.state_dict(), 'best_model.pth')
    else:
        trigger_times += 1
        if trigger_times >= patience:
            print('Early stopping!')
            break

# --- 6. Load Best Model and Generate Test Predictions ---
model.load_state_dict(torch.load('best_model.pth'))
model.eval()
test_preds = []

with torch.no_grad():
    for data in test_loader:
        out = model(data)
        pred = out.argmax(dim=1)
        test_preds.extend(pred.numpy())

# --- 7. Write Submission File ---
submission_df = pd.DataFrame({
    'id': [graph.graph_id for graph in test_graphs],
    'target': test_preds
})

submission_df.to_csv('./submissions/inbox/submission.csv', index=False)
print("Submission file generated successfully.")
